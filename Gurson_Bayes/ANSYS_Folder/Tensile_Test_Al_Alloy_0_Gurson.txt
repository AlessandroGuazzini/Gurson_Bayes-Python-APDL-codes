! >>>>> CONSTANTS <<<<<<
pigreco = acos(-1)

! >>>>> GEOMETRY <<<<<
GL = 25                       ! gauge length (mm)

! >>>>> MATERIAL PROPERTIES <<<<<
EYOUNG = 69279                ! Young's modulus (MPa) eng. σ-ε
ni = 0.33

! >>>>> MESH PARAMETERS <<<<< 
NDIVR = 40                     ! n° of elements along the radial direction 
NDIVL = 80                     ! n° of elements along the longitudinal direction 

! >>>>> LOADING PARAMETERS <<<<<
Umax = 1.7                     ! (mm)
NLST = 50                      ! Loadstep number

! >>>>> GTN MODEL PARAMETERS <<<<<
f0 = 0.0			! initial porosity
q1 = 1.5			! first Tvergaard constant
q2 = 1.0			! second Tvergaard constant
q3 = 2.25			! third Tvergaard constant = q1^2   

!i valori di q1,q2 e q3 sono fissati dal modello di Gurson

fN = 0.06			! nucleation porosity
epsN = 0.2			! mean strain
sN = 0.05			! standard deviation of mean strain
sigmaY = 504.337561157479  ! yield stress

gursact = 1			! activate Gurson model

/PREP7
K, 1, 0, 0, 0

K, 4, 0, 0.5*GL, 0

x0 = 0.5*SD-Delta
y0 = 0
x1 = 0.5*SD
y1 = 0.5*GL
a1 = ((x0**2+y0**2)-(x1**2+y1**2))/Delta
a2 = -a1*x0-(x0**2+y0**2)
Radius = (a1**2/4-a2)**0.5

CLOCAL, 11, 1, 0.5*SD-Delta+Radius, 0, 0, 0, 0, 0
L, 2, 3 $ CSYS, 0 $ LESIZE, 1, , , NDIVL, 1
L, 3, 4 $ LESIZE, 2, , , NDIVR, 1
L, 4, 1 $ LESIZE, 3, , , NDIVL, 1
L, 1, 2 $ LESIZE, 4, , , NDIVR, 1

A, 1, 2, 3, 4

!TB, MISO, 1, 1, 5

! Gurson (gursact different than 0)
*IF, gursact, NE, 0, THEN
  ! GTN base model
  TB, GURSON, 1, , 5, BASE
  TBDATA, 1, sigmaY, f0, q1, q2, q3
  ! GTN Strain-controlled nucleation
  TB, GURSON, 1, , 3, SNNU
  TBDATA, 1, fN, epsN, sN
