! -------------------------------------------------------------------------
! Uniaxial V-notched Tensile Test FE Modelling
! -------------------------------------------------------------------------

FINISH
/CLEAR, START, NEW
/FILNAME,VTT01,1
/DSCAL, ALL, 1

! >>>>> CONSTANTS <<<<<
pi = acos(-1)
eps = 1E-04

*DIM,MyTitle,STRING,22
MyTitle(1) = STRFILL(MyTitle,'VTT_FullResults_01',1)


! >>>>> GEOMETRIC PARAMETERS <<<<<
l = 50
gl = 25
w = 10
d = 5.85
r = 0.2
h = 0.43
tk = 1.52
deltar = d/6
alpha = 58.4*(pi/180)

! >>>>> MATERIAL PARAMETERS <<<<<
EYoung = 208000               ! Young modulus (MPa)
ni = 0.3                             ! Poisson ratio
eps_kill = 1.0

! >>>>> GTN MODEL PARAMETERS <<<<<
f0 = 0.0			! initial porosity
q1 = 1.5			! first Tvergaard constant
q2 = 1.0			! second Tvergaard constant
q3 = 2.25			! third Tvergaard constant = q1^2    
fN = 0.0352			!!! nucleation porosity
sN = 0.05			! standard deviation of mean strain
epsN = 0.0923			!!! mean strain
sigmaY = 810.714150		! initial yielding strength (MPa)

! >>>>> LOADING CONDITIONS <<<<<
uf = 0.5			! total displacement
NLSTLC = 1000	    ! n° of the load steps (500 NLSTC works), otherwise 1000*uf
killact = 0			! killing activation (if different than 0)
gursact = 1			! gurson activation (if different than 0)

! >>>>> MESH PARAMETERS <<<<<
meshfact = 3.0          !3.0 works
rxdiv = 8*meshfact 
rydiv = 5*meshfact 
nhdiv = NINT(rydiv*h/deltar) 
n1div = 6*meshfact 
n2div = 3*meshfact 
n3div = NINT(n2div*(gl-w)/w) 
n4div = NINT(n2div*(l-gl)/w) 
ntkdiv = 4*meshfact 

/PREP7
! >>>>> SPECIMEN MODELING <<<<<
ET, 1, SOLID186

MP, EX, 1, EYoung
MP, PRXY, 1, ni

! Gurson (gursact different than 0)
*IF, gursact, NE, 0, THEN
  ! GTN base model
  TB, GURSON, 1, , 5, BASE
  TBDATA, 1, sigmaY, f0, q1, q2, q3
  ! GTN Strain-controlled nucleation
  TB, GURSON, 1, , 3, SNNU
  TBDATA, 1, fN, epsN, sN
  ! GTN Coalescence
  !TB, GURSON, 1, , 2, COAL
  !TBDATA, 1, fc, fF
*ENDIF

TB, PLASTIC, 1, 1, 5270, MISO 
/INPUT, Costitutiva, txt

K, 1, 0, 0
K, 2, d/2-deltar, 0
K, 3, d/2, 0
K, 4, 0, h/2
K, 5, d/2-deltar, h/2
K, 6, d/2, h/2
K, 7, d/2+r, h/2
K, 8, d/2+r*(1-SIN(alpha/2)), r*COS(alpha/2)+h/2
K, 9, w/2, r*COS(alpha/2)+(w/2-d/2-r*(1-SIN(alpha/2)))*TAN(alpha/2)+h/2
K, 10, d/2+r-(r+deltar)*SIN(alpha/2), (r+deltar)*COS(alpha/2)+h/2
K, 11, w/2, (r+deltar)*COS(alpha/2)+(w/2-d/2-r+(r+deltar)*SIN(alpha/2))*TAN(alpha/2)+h/2
K, 12, w/2, w/2
K, 13, 0, w/2
K, 14, w/2, gl/2
K, 15, 0, gl/2
K, 16, w/2, l/2
K, 17, 0, l/2
K, 18, 0, 0, tk/2

L, 1, 2                 $ LESIZE, 1, , , n1div
L, 2, 3                 $ LESIZE, 2, , , rxdiv
L, 3, 6                 $ LESIZE, 3, , , nhdiv
L, 1, 4                 $ LESIZE, 4, , , nhdiv 
L, 2, 5                 $ LESIZE, 5, , , nhdiv 
L, 4, 5                 $ LESIZE, 6, , , n1div
L, 5, 6                 $ LESIZE, 7, , , rxdiv 
LARC, 6, 8, 7, r        $ LESIZE, 8, , , rydiv 
LARC, 5, 10, 7, r+deltar $ LESIZE, 9, , , rydiv 
L, 8, 10                 $ LESIZE, 10, , , rxdiv 
L, 8, 9                 $ LESIZE, 11, , , n2div
L, 9, 11                 $ LESIZE, 12, , , rxdiv 
L, 10, 11                 $ LESIZE, 13, , , n2div
L, 4, 13                $ LESIZE, 14, , , rydiv 
L, 10, 13                $ LESIZE, 15, , , n1div
L, 11, 12                 $ LESIZE, 16, , , n1div
L, 13, 12                $ LESIZE, 17, , , n2div
L, 12, 14                $ LESIZE, 18, , , n3div
L, 14, 15               $ LESIZE, 19, , , n2div
L, 13, 15               $ LESIZE, 20, , , n3div
L, 14, 16               $ LESIZE, 21, , , n4div
L, 16, 17               $ LESIZE, 22, , , n2div
L, 15, 17               $ LESIZE, 23, , , n4div
L, 1, 18                $ LESIZE, 24, , , ntkdiv

AL, 2, 3, 7, 5
AL, 1, 5, 6, 4
AL, 7, 8, 10, 9
AL, 6, 9, 15, 14
AL, 11, 12, 13, 10
AL, 13, 16, 17, 15
AL, 17, 18, 19, 20
AL, 19, 21, 22, 23

VDRAG, ALL, , , , , , 24

TYPE, 1 $ MAT, 1 $ MSHKEY, 1 $ MSHAPE, 0
VMESH, all

NLGEOM, ON

FINISH

! >>>>> SOLUTION <<<<<
/SOLU
ANTYPE, STATIC, NEW
NROPT, FULL, , OFF
EQSLV, SPARSE
KBC, 0
NCNV, 0
AUTOTS, ON
NSUBST, 1, 1000, 1

NSEL, S, LOC, X, -eps, eps
D, ALL, UX, 0 
NSEL, S, LOC, Y, -eps, eps
D, ALL, UY, 0
NSEL, S, LOC, Z, -eps, eps
D, ALL, UZ, 0
NSEL, ALL

tstep = 1

*DO, i, 1, nlstlc, 1
  TIME, tstep
  ASEL, S, AREA, , 37 $ NSLA, S, 1
  D, ALL, UY, (uf/(2*nlstlc))*i $ ASEL, ALL $ NSEL, ALL
  SOLVE
  tstep = tstep + 1

  ! killing (killact different than 0)
  *IF, killact, NE, 0, THEN
    ! selection of elem to kill
    ESEL, S, LIVE
    NSLE, S, CORNER
    NSEL, R, EPTO, 1, eps_kill, 10*eps_kill
    ESLN, S, 0, CORNER ! (if = 1) select elem with all of their nodes selected
    ! element killing one-by-one
    *GET, numelem, ELEM, 0, COUNT
    *IF, numelem, GT, 0, THEN
      *VGET, elemlist, ELEM, , ELIST
      ALLS
      *DO, j, 1, numelem, 1
        TIME, tstep
        EKILL, elemlist(j)
        SOLVE
        tstep = tstep + 1
      *ENDDO
      elemlist =
      *EXIT
    *ENDIF
    ALLS
  *ENDIF

*ENDDO


FINISH

SAVE



/eof
! >>>>> RESULTS <<<<<
/POST1

! PLESOL, DPAR, TPOR, 2

/OUTPUT, MT_Vnotch, txt, , NEW
!/OUTPUT
*GET, nlstlc, ACTIVE, 0, SOLU, NCMLS ! nlst?
*DO, i, 1, nlstlc, 1
 SET, i
 
 KSEL, S, KP, , 14 $ NSLK, S
 *GET, nmrnd, NODE, , NUM, MAX $ ALLS
 *GET, uy, NODE, NMRND, U, Y
 uy = 2*uy

!! Define reaction force
 ASEL, S, AREA, , 37 
 NSLA, S, 1 
 FSUM
 *GET, ry, FSUM, 0, ITEM, FY $ ASEL, ALL $ NSEL, ALL
 
 !! Second variation if ASEL doesn't work
 !NSEL, S, LOC, Y, 0
 !FSUM
 !*GET, ry, FSUM, 0, ITEM, FY
 !NSEL, ALL

 Frc = -4*ry

 /OUTPUT, MT_Vnotch, txt, , APPEND
 *VWRITE, uy, Frc
 (2X,F10.5,2X,F10.3)
 /OUTPUT
*ENDDO

FINISH
